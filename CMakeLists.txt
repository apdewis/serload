cmake_minimum_required(VERSION 3.10)
project(serload)
include_directories(../argparse)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SERIALPORT REQUIRED IMPORTED_TARGET libserialport)
add_executable(${PROJECT_NAME} serload.c argparse.c)
target_link_libraries(${PROJECT_NAME} m PkgConfig::SERIALPORT)
install(TARGETS ${PROJECT_NAME} DESTINATION host/bin)
